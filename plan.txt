_clip_to_face_roi(roi) -> BoundingBox | None :
def _clip_to_face_roi(self, roi):
    if roi is None or not self.face_priority_mode:
        return roi
    mode = getattr(self.cfg, 'face_clip_mode', 'none')
    if mode == 'none':
        return roi
    rx, ry, rw, rh = roi
    if mode == 'closeup':
        # Bbox IS the face: skip top 25% hair, keep 35% eye zone
        skip = int(rh * 0.25)
        h    = max(8, int(rh * 0.35))
        return (rx, ry + skip, rw, h)
    if mode == 'full_body_head':
        # Extract head from top of body bbox, then center on eye zone
        head_frac = getattr(self.cfg, 'face_head_frac', 0.22)
        eye_off   = getattr(self.cfg, 'face_eye_offset', 0.48)
        head_h    = max(8, int(rh * head_frac))
        # Hair skip = top 20% of head (less aggressive than closeup)
        skip      = int(head_h * 0.20)
        # Eye band = centered at eye_off fraction of head
        eye_top   = int(head_h * (eye_off - 0.20))
        eye_h     = max(8, int(head_h * 0.40))
        return (rx, ry + skip + eye_top, rw, eye_h)
    return roi
